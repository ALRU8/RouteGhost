#!/usr/bin/env python3
"""Install the RouteGhost bridge into an upstream Organic Maps source tree.

Usage: python apply_patch.py /path/to/organicmaps
"""
from pathlib import Path
import shutil
import re
import sys
import xml.etree.ElementTree as ET

OVERLAY = Path(__file__).resolve().parent / "overlay" / "app" / "organicmaps" / "routeghost"


def once_replace(data: str, before: str, after: str, location: str) -> str:
    number = data.count(before)
    if number != 1:
        raise RuntimeError(f"Organic Maps version mismatch in {location}: expected 1 anchor, got {number}: {before[:90]}")
    return data.replace(before, after, 1)


def set_routeghost_label(xml: str, tag_name: str, identity: str = "") -> str:
    """Rename only a specific manifest element, preserving the original formatting.

    Organic Maps defines the app name on both <application> and its launcher
    <activity-alias>. Replacing a global string or expecting one match breaks.
    """
    matches = [match for match in re.finditer(r"<" + re.escape(tag_name) + r"\b[^>]*>", xml, re.DOTALL)
               if not identity or identity in match.group(0)]
    if len(matches) != 1:
        raise RuntimeError(f"Expected exactly one <{tag_name}> element containing {identity!r}; got {len(matches)}")
    match = matches[0]
    tag = match.group(0)
    if 'android:label="RouteGhost"' in tag:
        return xml
    before = 'android:label="@string/app_name"'
    if tag.count(before) != 1:
        raise RuntimeError(f"Unexpected label in <{tag_name}> with {identity!r}")
    updated = tag.replace(before, 'android:label="RouteGhost"', 1)
    return xml[:match.start()] + updated + xml[match.end():]


def patch_native_route_geometry(native_text: str) -> str:
    # Patch just one JNI function. Tolerate upstream whitespace and logging changes.
    name = "Java_app_organicmaps_sdk_Framework_nativeGetRouteJunctionPoints"
    matches = list(re.finditer(r"\b" + name + r"\s*\(", native_text))
    if len(matches) != 1:
        raise RuntimeError(f"Expected exactly one {name} in Framework.cpp; got {len(matches)}")
    opening = native_text.find("{", matches[0].end())
    if opening == -1:
        raise RuntimeError("No opening brace in nativeGetRouteJunctionPoints")

    depth = 0
    closing = -1
    for pos in range(opening, len(native_text)):
        if native_text[pos] == "{":
            depth += 1
        elif native_text[pos] == "}":
            depth -= 1
            if depth == 0:
                closing = pos
                break
    if closing < 0:
        raise RuntimeError("Unbalanced JNI method braces")

    body = native_text[opening + 1:closing]
    if "Selected route has no usable geometry" in body and "session.RouteCall" in body:
        return native_text

    start = re.search(r"std::vector\s*<\s*geometry::PointWithAltitude\s*>\s+points\s*;", body)
    finish = re.search(r"std::vector\s*<\s*geometry::PointWithAltitude\s*>\s+result\s*;", body)
    if not start or not finish or finish.start() <= start.end():
        raise RuntimeError("Unexpected JNI geometry function: missing points/result declarations")
    original = body[start.end():finish.start()]
    if "GetRouteJunctionPoints(points)" not in original:
        raise RuntimeError("Unexpected JNI geometry method: routing source differs")
    if "result.push_back(points[0])" not in body[finish.end():]:
        raise RuntimeError("Unexpected interpolation code; aborting")

    new_part = """std::vector<geometry::PointWithAltitude> points;
  auto const & session = frm()->GetRoutingManager().RoutingSession();
  if (!session.GetRouteJunctionPoints(points) || points.size() < 2)
  {
    points.clear();
    session.RouteCall([&points](routing::RoutesResult const & routes)
    {
      if (!routes.IsValid())
        return;
      routes.GetActive().ForEachPoint([&points](geometry::PointWithAltitude const & point)
      {
        points.push_back(point);
      });
    });
  }
  if (points.size() < 2)
  {
    LOG(LWARNING, ("Selected route has no usable geometry"));
    return nullptr;
  }
  std::vector<geometry::PointWithAltitude> result;"""
    body = body[:start.start()] + new_part + body[finish.end():]
    return native_text[:opening + 1] + body + native_text[closing:]


def patch(root: Path):
    main = root / "android" / "app" / "src" / "main"
    mwm = main / "java" / "app" / "organicmaps" / "MwmActivity.java"
    manifest = main / "AndroidManifest.xml"
    if not mwm.is_file() or not manifest.is_file():
        raise FileNotFoundError("Expected original Organic Maps checkout (android/app/src/main/...)")
    text = mwm.read_text(encoding="utf-8")
    hook = "    initViews(isLaunchByDeepLink);"
    marker = "app.organicmaps.routeghost.RouteGhostUi.attach(this);"
    if marker not in text:
        text = once_replace(text, hook, hook + "\n    " + marker, "MwmActivity.java")
    mwm.write_text(text, encoding="utf-8")

    xml = manifest.read_text(encoding="utf-8")
    if 'android:name="app.organicmaps.routeghost.RouteGhostPanel"' not in xml:
        insertion = '''    <!-- RouteGhost: integrated test-location controls. -->
    <activity
        android:name="app.organicmaps.routeghost.RouteGhostPanel"
        android:label="RouteGhost"
        android:exported="false"
        android:theme="@android:style/Theme.Material.Light.NoActionBar" />
    <service
        android:name="app.organicmaps.routeghost.LocationService"
        android:exported="false"
        android:foregroundServiceType="location" />
'''
        xml = once_replace(xml, "</application>", insertion + "  </application>", "AndroidManifest.xml")
    xml = set_routeghost_label(xml, "application")
    xml = set_routeghost_label(xml, "activity-alias",
                                'android:name="app.organicmaps.DownloadResourcesActivity"')
    # The upstream app already requests ACCESS_FINE_LOCATION, FOREGROUND_SERVICE and FOREGROUND_SERVICE_LOCATION.
    xmlroot = ET.fromstring(xml)
    ns = "{http://schemas.android.com/apk/res/android}"
    permission_names = {node.attrib.get(ns + "name") for node in xmlroot if node.tag == "uses-permission"}
    for name in ["android.permission.ACCESS_FINE_LOCATION", "android.permission.FOREGROUND_SERVICE", "android.permission.FOREGROUND_SERVICE_LOCATION"]:
        if name not in permission_names:
            raise RuntimeError(f"Organic Maps manifest is missing required permission {name}")
    manifest.write_text(xml, encoding="utf-8")

    # When the route is previewed or the selected alternative has just switched,
    # the native following-route geometry may not be ready yet. The routing core
    # retains the selected route in RoutesResult, including its actual vertices.
    # Read these vertices as a fallback instead of inventing a straight line.
    native_routing = (root / "android" / "sdk" / "src" / "main" / "cpp" / "app" /
                      "organicmaps" / "sdk" / "Framework.cpp")
    if not native_routing.is_file():
        raise FileNotFoundError("Missing upstream SDK JNI Framework.cpp")
    native_text = native_routing.read_text(encoding="utf-8")
    patched_text = patch_native_route_geometry(native_text)
    if patched_text != native_text:
        native_routing.write_text(patched_text, encoding="utf-8")

    destination = main / "java" / "app" / "organicmaps" / "routeghost"
    destination.mkdir(parents=True, exist_ok=True)
    for f in OVERLAY.glob("*.java"):
        shutil.copyfile(f, destination / f.name)
    print(f"RouteGhost integrated into {root}")
    print(f"Java classes: {len(list(destination.glob('*.java')))}")


if __name__ == "__main__":
    if len(sys.argv) != 2:
        raise SystemExit("Usage: python apply_patch.py /path/to/organicmaps")
    patch(Path(sys.argv[1]).resolve())

package app.organicmaps.routeghost;

/** Developer-options mock location only. Does not hide or spoof the mock flag. */
public final class LocationService extends Service {
    public static final String START="app.organicmaps.routeghost.START", STOP="app.organicmaps.routeghost.STOP", PAUSE="app.organicmaps.routeghost.PAUSE", RESUME="app.organicmaps.routeghost.RESUME", UPDATE="app.organicmaps.routeghost.UPDATE";
    public static volatile boolean running=false, paused=false, fixed=false;
    public static volatile String status="Эмуляция выключена";
    public static volatile Pt position=null;
    public static volatile double speed=0;
    public static volatile long elapsed=0,total=0;
    private static final String CHANNEL="routeghost_simulation";
    private final Handler handler=new Handler(Looper.getMainLooper());
    private final ArrayList<String> providers=new ArrayList<>();
    private LocationManager manager;
    private PowerManager.WakeLock wake;
    private Timeline timeline;
    private Pt fixedPoint;
    private boolean noisy,subwayLoss,loop,autoStop;
    private long startedAt, pausedAt, pausedDuration, seed;
    private float multiplier=1f;
    private long lastNotification=0;

    @Override public IBinder onBind(Intent intent){return null;}
    @Override public int onStartCommand(Intent i,int flags,int startId){
        if(i==null)return START_NOT_STICKY;
        String action=i.getAction();
        if(STOP.equals(action)){halt("Эмуляция остановлена");return START_NOT_STICKY;}
        if(PAUSE.equals(action)){if(running && !paused){paused=true;pausedAt=SystemClock.elapsedRealtime();status="Пауза";notifyStatus();}return START_NOT_STICKY;}
        if(RESUME.equals(action)){if(running && paused){pausedDuration+=SystemClock.elapsedRealtime()-pausedAt;paused=false;status="Продолжается";notifyStatus();}return START_NOT_STICKY;}
        if(!START.equals(action))return START_NOT_STICKY;
        if(running){
            handler.removeCallbacks(tick);
            if(manager!=null)for(String provider:providers){try{manager.removeTestProvider(provider);}catch(Exception ignored){}}
            providers.clear();
            if(wake!=null&&wake.isHeld())wake.release();wake=null;
            running=false;paused=false;
        }
        try{
            makeForeground("Запуск GPS-тестирования");
            fixed=i.getBooleanExtra("fixed",false);
            noisy=i.getBooleanExtra("noise",false);subwayLoss=i.getBooleanExtra("subway_loss",false);
            loop=i.getBooleanExtra("loop",false);autoStop=i.getBooleanExtra("auto_stop",false);
            seed=i.getLongExtra("seed",System.currentTimeMillis());
            multiplier=Math.max(0.25f,Math.min(10f,i.getFloatExtra("multiplier",1)));
            if(fixed){fixedPoint=new Pt(i.getDoubleExtra("lat",55.7512),i.getDoubleExtra("lon",37.6184));total=0;timeline=null;}
            else{
                File input=new File(getFilesDir(),"active-timeline.json");
                if(!input.exists()||input.length()>20_000_000)throw new Exception("Сценарий маршрута не найден или слишком большой");
                byte[] bytes=new byte[(int)input.length()];try(FileInputStream in=new FileInputStream(input)){int off=0;while(off<bytes.length){int n=in.read(bytes,off,bytes.length-off);if(n==-1)break;off+=n;}}
                timeline=Timeline.read(new JSONObject(new String(bytes,StandardCharsets.UTF_8)));
                if(timeline.frames.isEmpty())throw new Exception("Маршрут пустой");
                total=timeline.duration();
            }
            manager=getSystemService(LocationManager.class);
            for(String p:new String[]{LocationManager.GPS_PROVIDER,LocationManager.NETWORK_PROVIDER}){
                try {
                    manager.addTestProvider(p,false,false,false,false,true,true,true,1,1);
                    manager.setTestProviderEnabled(p,true);providers.add(p);
                }catch(SecurityException ex){if(providers.isEmpty())throw ex;}
                catch(IllegalArgumentException ex){ /* Provider conflict: GPS is mandatory. */ if(providers.isEmpty())throw ex; }
            }
            if(providers.isEmpty())throw new Exception("Не удалось подключить тестовые GPS-провайдеры");
            PowerManager powers=getSystemService(PowerManager.class);
            wake=powers.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK,"RouteGhost:GPSReplay");wake.acquire();
            startedAt=SystemClock.elapsedRealtime();pausedDuration=0;paused=false;running=true;status=fixed?"Точка закреплена":"Движение по маршруту";
            tick.run();
        }catch(SecurityException e){halt("Выберите RouteGhost в настройках разработчика → фиктивные местоположения");}
        catch(Exception e){halt("Ошибка запуска: "+e.getMessage());}
        return START_NOT_STICKY;
    }
    private final Runnable tick=new Runnable(){@Override public void run(){
        if(!running)return;
        try{
            long now=SystemClock.elapsedRealtime();
            long ms=paused?(pausedAt-startedAt-pausedDuration):(now-startedAt-pausedDuration);
            long current=Math.max(0,(long)(ms*multiplier));
            elapsed=fixed?0:loop&&total>0?current%total:Math.min(current,total);
            Sample sample;
            if(fixed){Pt p=fixedPoint;
                if(noisy){double phase=now/1000.0;double dy=1.1*Math.sin(phase*0.21+seed*.001);double dx=1.0*Math.cos(phase*.17+seed*.001);
                    p=new Pt(p.lat+dy/111111.0,p.lon+dx/(111111.0*Math.max(0.1,Math.cos(Math.toRadians(p.lat)))));
                }
                sample=new Sample(p,0,0,"Фиксированная точка",WAIT);
            }else{sample=SimulationEngine.sample(timeline,elapsed,noisy,seed);}
            position=sample.pos;speed=paused?0:sample.speedMps*(fixed?0:multiplier);
            status=paused?"Пауза":fixed?"Фиксированная точка":elapsed>=total?"Маршрут завершён, точка удерживается":sample.label;
            // Underground subway tunnels often have no satellite position. Optionally
            // retain the last provider fix until the train leaves a tunnel.
            if(!(subwayLoss && METRO.equals(sample.type) && !fixed))for(String provider:providers){
                Location loc=new Location(provider);
                loc.setLatitude(sample.pos.lat);loc.setLongitude(sample.pos.lon);
                loc.setAccuracy(noisy?5f:3f);loc.setSpeed((float)speed);loc.setBearing((float)sample.bearing);
                loc.setTime(System.currentTimeMillis());loc.setElapsedRealtimeNanos(SystemClock.elapsedRealtimeNanos());
                manager.setTestProviderLocation(provider,loc);
            }
            if(now-lastNotification>8000){notifyManager();lastNotification=now;}
            sendBroadcast(new Intent(UPDATE).setPackage(getPackageName()));
            if(!fixed&&!loop&&autoStop&&current>=total){handler.postDelayed(()->halt("Маршрут завершён"),1000);}
            else handler.postDelayed(this,1000);
        }catch(Exception e){halt("Ошибка GPS: "+e.getMessage());}
    }};
    private void makeForeground(String desc){
        NotificationManager nm=getSystemService(NotificationManager.class);
        nm.createNotificationChannel(new NotificationChannel(CHANNEL,"Эмуляция GPS",NotificationManager.IMPORTANCE_LOW));
        startForeground(1701,notification(desc));
    }
    private Notification notification(String desc){
        int flags=PendingIntent.FLAG_IMMUTABLE|PendingIntent.FLAG_UPDATE_CURRENT;
        PendingIntent open=PendingIntent.getActivity(this,0,new Intent(this,RouteGhostPanel.class),flags);
        PendingIntent stop=PendingIntent.getService(this,1,new Intent(this,LocationService.class).setAction(STOP),flags);
        PendingIntent pause=PendingIntent.getService(this,2,new Intent(this,LocationService.class).setAction(paused?RESUME:PAUSE),flags);
        return new Notification.Builder(this,CHANNEL)
            .setSmallIcon(android.R.drawable.ic_menu_mylocation).setContentTitle("RouteGhost · тестовый GPS")
            .setContentText(desc).setContentIntent(open).setOngoing(true)
            .addAction(new Notification.Action.Builder(null,paused?"Продолжить":"Пауза",pause).build())
            .addAction(new Notification.Action.Builder(null,"Стоп",stop).build()).build();
    }
    private void notifyManager(){getSystemService(NotificationManager.class).notify(1701,notification(status));}
    private void notifyStatus(){notifyManager();sendBroadcast(new Intent(UPDATE).setPackage(getPackageName()));}
    private void halt(String desc){
        handler.removeCallbacks(tick);
        running=false;paused=false;position=null;speed=0;elapsed=0;total=0;
        if(manager!=null)for(String provider:providers){try{manager.removeTestProvider(provider);}catch(Exception ignored){}}
        providers.clear();
        if(wake!=null&&wake.isHeld())wake.release();wake=null;
        status=desc;
        sendBroadcast(new Intent(UPDATE).setPackage(getPackageName()));
        try{stopForeground(STOP_FOREGROUND_REMOVE);}catch(Exception ignored){}
        stopSelf();
    }
    @Override public void onDestroy(){if(running)halt("Эмуляция выключена");super.onDestroy();}
}

package app.organicmaps.routeghost;

import org.json.JSONArray;
import org.json.JSONException;
import org.json.JSONObject;
import java.util.ArrayList;
import java.util.List;
import java.util.Locale;

public final class RouteData {
    private RouteData() {}
    public static final String WALK="WALK", CAR="CAR", BUS="BUS", ELECTROBUS="ELECTROBUS", MINIBUS="MINIBUS", TROLLEY="TROLLEY", TRAM="TRAM",
      METRO="METRO", MCC="MCC", MCD="MCD", TRAIN="TRAIN", MONORAIL="MONORAIL", FERRY="FERRY", OTHER="OTHER", WAIT="WAIT";

    public static final class Pt {
        public final double lat, lon;
        public Pt(double lat, double lon) { if (!Double.isFinite(lat)||!Double.isFinite(lon)||Math.abs(lat)>90||Math.abs(lon)>180) throw new IllegalArgumentException("Неверные координаты"); this.lat=lat;this.lon=lon; }
        public JSONArray json() throws JSONException { JSONArray a=new JSONArray(); a.put(lat);a.put(lon);return a; }
        public static Pt read(JSONArray a) throws JSONException {return new Pt(a.getDouble(0),a.getDouble(1));}
        @Override public String toString(){return String.format(Locale.US,"%.6f, %.6f",lat,lon);}
    }
    public static class Leg {
        public String type, name;
        public final ArrayList<Pt> path = new ArrayList<>();
        public final ArrayList<Pt> stops = new ArrayList<>();
        public int baseSeconds;
        public boolean inferredGeometry=false;
        public Leg(String type, String name, List<Pt> path, int seconds){this.type=type;this.name=name;this.path.addAll(path);baseSeconds=Math.max(1,seconds);}
        public JSONObject json() throws JSONException {
            JSONObject o=new JSONObject();o.put("type",type);o.put("name",name);o.put("seconds",baseSeconds);o.put("inferred",inferredGeometry);
            JSONArray arr=new JSONArray();for(Pt p:path)arr.put(p.json());o.put("path",arr);
            JSONArray ss=new JSONArray();for(Pt p:stops)ss.put(p.json());o.put("stops",ss);return o;
        }
        public static Leg read(JSONObject o) throws JSONException {
            ArrayList<Pt> path=new ArrayList<>(); JSONArray a=o.getJSONArray("path");for(int i=0;i<a.length();i++)path.add(Pt.read(a.getJSONArray(i)));
            Leg l=new Leg(o.getString("type"),o.optString("name",""),path,o.optInt("seconds",1));
            l.inferredGeometry=o.optBoolean("inferred",false);
            JSONArray ss=o.optJSONArray("stops");if(ss!=null) for(int i=0;i<ss.length();i++)l.stops.add(Pt.read(ss.getJSONArray(i)));
            return l;
        }
    }
    public static final class Plan {
        public String title="Маршрут", source="Локальный", note="";
        public final ArrayList<Leg> legs = new ArrayList<>();
        public long seed=System.currentTimeMillis();
        public JSONObject json() throws JSONException {
            JSONObject o=new JSONObject();o.put("title",title);o.put("source",source);o.put("note",note);o.put("seed",seed);
            JSONArray a=new JSONArray();for(Leg leg:legs)a.put(leg.json());o.put("legs",a);return o;
        }
        public static Plan read(JSONObject o) throws JSONException {
            Plan p=new Plan();p.title=o.optString("title","Маршрут");p.source=o.optString("source","");p.note=o.optString("note","");p.seed=o.optLong("seed",System.currentTimeMillis());
            JSONArray a=o.getJSONArray("legs");for(int i=0;i<a.length();i++)p.legs.add(Leg.read(a.getJSONObject(i)));return p;
        }
        public int totalBaseSeconds(){int n=0;for(Leg l:legs)n+=l.baseSeconds;return n;}
        public double length(){double d=0;for(Leg l:legs)d+=RouteData.length(l.path);return d;}
    }
    public static final class Keyframe {
        public final long atMillis;
        public final Pt pos;
        public final String label, type;
        public Keyframe(long atMillis, Pt pos, String label, String type){this.atMillis=atMillis;this.pos=pos;this.label=label;this.type=type;}
        public JSONObject json() throws JSONException {JSONObject o=new JSONObject();o.put("at",atMillis);o.put("pos",pos.json());o.put("label",label);o.put("type",type);return o;}
        public static Keyframe read(JSONObject o)throws JSONException{return new Keyframe(o.getLong("at"),Pt.read(o.getJSONArray("pos")),o.getString("label"),o.getString("type"));}
    }
    public static final class Timeline {
        public final ArrayList<Keyframe> frames=new ArrayList<>();
        public final ArrayList<String> events=new ArrayList<>();
        public long duration(){return frames.isEmpty()?0:frames.get(frames.size()-1).atMillis;}
        public JSONObject json()throws JSONException{JSONArray a=new JSONArray();for(Keyframe f:frames)a.put(f.json());JSONObject o=new JSONObject();o.put("frames",a);return o;}
        public static Timeline read(JSONObject o)throws JSONException {Timeline t=new Timeline();JSONArray a=o.getJSONArray("frames");for(int i=0;i<a.length();i++)t.frames.add(Keyframe.read(a.getJSONObject(i)));return t;}
    }
    public static final class Sample {
        public final Pt pos;
        public final double speedMps, bearing;
        public final String label,type;
        public Sample(Pt p,double s,double b,String label,String type){pos=p;speedMps=s;bearing=b;this.label=label;this.type=type;}
    }
    public static double metres(Pt a,Pt b){double r=6371000.0;double x=Math.toRadians(b.lon-a.lon)*Math.cos(Math.toRadians((a.lat+b.lat)/2));double y=Math.toRadians(b.lat-a.lat);return Math.sqrt(x*x+y*y)*r;}
    public static double length(List<Pt> path){double d=0;for(int i=1;i<path.size();i++)d+=metres(path.get(i-1),path.get(i));return d;}
    public static Pt mix(Pt a,Pt b,double q){q=Math.max(0,Math.min(1,q));return new Pt(a.lat+(b.lat-a.lat)*q,a.lon+(b.lon-a.lon)*q);}
    public static double bearing(Pt a,Pt b){double phi1=Math.toRadians(a.lat),phi2=Math.toRadians(b.lat),lambda=Math.toRadians(b.lon-a.lon);return (360+Math.toDegrees(Math.atan2(Math.sin(lambda)*Math.cos(phi2),Math.cos(phi1)*Math.sin(phi2)-Math.sin(phi1)*Math.cos(phi2)*Math.cos(lambda))))%360;}
    public static String display(String type){switch(type){case WALK:return "Пешком";case CAR:return "Автомобиль";case BUS:return "Автобус";case ELECTROBUS:return "Электробус";case MINIBUS:return "Маршрутка";case TROLLEY:return "Троллейбус";case TRAM:return "Трамвай";case METRO:return "Метро";case MCC:return "МЦК";case MCD:return "МЦД";case TRAIN:return "Электричка";case MONORAIL:return "Монорельс";case FERRY:return "Водный транспорт";case WAIT:return "Ожидание";default:return "Транспорт";}}
    public static boolean rail(String mode){return mode.equals(METRO)||mode.equals(MCC)||mode.equals(MCD)||mode.equals(TRAIN)||mode.equals(MONORAIL);}
    public static boolean publicTransport(String type){return !(WALK.equals(type)||CAR.equals(type)||WAIT.equals(type));}
}

    private Plan parseCustom(String source)throws Exception{
        String[] lines=source.trim().split("\\r?\\n");
        if(lines.length<2||lines.length>10000)throw new Exception("Укажите от 2 до 10000 точек");
        Plan plan=new Plan();plan.title="Свой маршрут";plan.source="Ручной ввод";
        Pt last=null;
        for(String line:lines){
            if(line.trim().isEmpty())continue;
            String[] parts=line.split(",");if(parts.length<2)throw new Exception("Строка должна содержать широту, долготу");
            Pt p=new Pt(Double.parseDouble(parts[0].trim()),Double.parseDouble(parts[1].trim()));
            String type=parts.length>2?parts[2].trim().toUpperCase(Locale.ROOT):WALK;
            if(!validType(type))throw new Exception("Неизвестный тип движения: "+type);
            if(last!=null){
                ArrayList<Pt> path=new ArrayList<>();path.add(last);path.add(p);
                double metres=metres(last,p);
                if(metres>100_000)throw new Exception("Слишком длинный разрыв между точками");
                int seconds=Math.max(1,(int)(metres/typicalSpeed(type)));
                plan.legs.add(new Leg(type,display(type),path,seconds));
            }
            last=p;
        }
        if(plan.legs.isEmpty())throw new Exception("Нужно минимум 2 точки");
        return plan;
    }

package app.organicmaps.routeghost;

import android.content.Context;
import org.json.JSONObject;
import java.io.*;
import java.nio.charset.StandardCharsets;
import static app.organicmaps.routeghost.RouteData.*;

public final class RouteGhostStorage {
    private RouteGhostStorage() {}
    private static File file(Context c){return new File(c.getFilesDir(),"routeghost-plan.json");}
    public static void save(Context c,Plan plan)throws Exception{
        byte[] bytes=plan.json().toString().getBytes(StandardCharsets.UTF_8);
        if(bytes.length>10_000_000)throw new Exception("Маршрут превышает допустимый размер");
        try(FileOutputStream out=new FileOutputStream(file(c))){out.write(bytes);}
    }
    public static Plan load(Context c)throws Exception{
        File f=file(c);
        if(!f.isFile()||f.length()>10_000_000)throw new Exception("Сначала постройте или импортируйте маршрут");
        byte[] data=new byte[(int)f.length()];
        try(InputStream in=new FileInputStream(f)){
            int offset=0,n;while(offset<data.length&&(n=in.read(data,offset,data.length-offset))!=-1)offset+=n;
            if(offset!=data.length)throw new IOException("Неполный файл маршрута");
        }
        return Plan.read(new JSONObject(new String(data,StandardCharsets.UTF_8)));
    }
    public static void saveTimeline(Context c,Timeline timeline)throws Exception{
        byte[] data=timeline.json().toString().getBytes(StandardCharsets.UTF_8);
        if(data.length>20_000_000)throw new Exception("Сценарий слишком большой");
        try(FileOutputStream out=new FileOutputStream(new File(c.getFilesDir(),"active-timeline.json"))){out.write(data);}
    }
}

    private static void captureWithRetry(Activity activity, TextView launch, Handler handler, int attempt) {
        if (activity.isFinishing() || (Build.VERSION.SDK_INT >= 17 && activity.isDestroyed())) return;

        RoutingController controller = RoutingController.get();
        boolean built = controller.isBuilt() || Framework.nativeIsRouteBuilt();
        boolean building = controller.isBuilding() || Framework.nativeIsRouteBuilding();
        boolean captured = false;
        String error = null;

        if (built && !building) {
            try {
                captured = captureSelectedRoute(activity);
            } catch (Exception ex) {
                error = ex.getMessage();
            }
        }

        if (!captured && attempt < MAX_ATTEMPTS && (building || built)) {
            handler.postDelayed(() -> captureWithRetry(activity, launch, handler, attempt + 1), RETRY_DELAY_MS);
            return;
        }

        if (error != null) {
            Toast.makeText(activity, "Не удалось прочитать маршрут: " + error, Toast.LENGTH_LONG).show();
        }
        launch.setText("GPS  ◉");
        launch.setEnabled(true);
        Intent intent = new Intent(activity, RouteGhostPanel.class);
        intent.putExtra("built", built || building);
        intent.putExtra("captured", captured);
        intent.putExtra("capture_attempted", true);
        activity.startActivity(intent);
    }

    private static int dp(Activity a, int value) {
        return (int) (value * a.getResources().getDisplayMetrics().density + 0.5f);
    }

    private static boolean captureSelectedRoute(Activity activity) throws Exception {
        // This JNI snapshot is tied to the selected alternative (not route marks).
        // The native fallback added by apply_patch.py can also read its stored RouteBase.
        JunctionInfo[] junctions = Framework.nativeGetRouteJunctionPoints(50.0);
        if (junctions == null || junctions.length < 2) return false;

        ArrayList<Pt> points = new ArrayList<>();
        for (JunctionInfo junction : junctions) {
            if (junction == null) continue;
            Pt point;
            try {
                point = new Pt(junction.mLat, junction.mLon);
            } catch (IllegalArgumentException ex) {
                throw new Exception("Маршрут содержит неверные координаты");
            }
            if (points.isEmpty() || metres(points.get(points.size() - 1), point) > 0.15) {
                points.add(point);
            }
            if (points.size() > 100000) throw new Exception("Слишком длинный маршрут");
        }
        if (points.size() < 2 || length(points) < 5) return false;

        Router router = Router.get();
        String type = router == Router.Vehicle ? CAR
            : router == Router.Pedestrian ? WALK
            : router == Router.Bicycle ? OTHER : BUS;
        int seconds = (int) Math.max(1, length(points) / (WALK.equals(type) ? 1.35 : 10.0));
        if (router == Router.Transit) {
            TransitRouteInfo info = RoutingController.get().getCachedTransitInfo();
            if (info != null) {
                seconds = Math.max(1, info.getTotalTime());
                for (TransitStepInfo step : info.getTransitSteps()) {
                    String t = step.getType().name();
                    if (t.equals("SUBWAY")) { type = METRO; break; }
                    if (t.equals("TRAM")) { type = TRAM; break; }
                    if (t.equals("TROLLEYBUS")) { type = TROLLEY; break; }
                    if (t.equals("TRAIN") || t.equals("LIGHT_RAIL")) { type = TRAIN; break; }
                    if (t.equals("BUS")) { type = BUS; break; }
                }
            }
        } else {
            RoutingInfo info = RoutingController.get().getCachedRoutingInfo();
            if (info != null && info.totalTimeInSeconds > 0) seconds = info.totalTimeInSeconds;
        }

        Plan plan = new Plan();
        plan.title = "Маршрут Organic Maps";
        plan.source = "Organic Maps";
        plan.note = router == Router.Transit
            ? "Геометрия выбранного варианта; скорость и пересадки моделируются приблизительно."
            : "";
        plan.legs.add(new Leg(type, "Маршрут", points, seconds));
        // Persist only after validating the entire selected path.
        RouteGhostStorage.save(activity, plan);
        return true;
    }
}

package app.organicmaps.routeghost;

import java.util.ArrayList;
import java.util.List;
import java.util.Random;
import static app.organicmaps.routeghost.RouteData.*;

/** Generates one reproducible journey. Never adds phantom geometry or invents real-world facts. */
public final class SimulationEngine {
    private SimulationEngine() {}
    public static final class Config {
        public boolean pauses=true, gpsNoise=true, trafficModel=true, subwaySignalLoss=false;
        public double pace=1.0;
        public int density=2; // 0 low, 1 standard, 2 realistic
    }
    private static int between(Random r,int lo,int hi){return lo+r.nextInt(Math.max(1,hi-lo+1));}
    private static boolean chance(Random r,double prob){return r.nextDouble()<prob;}
    private static int weighted(Random r,int[] values,int[] weights){int total=0;for(int w:weights)total+=w;int n=r.nextInt(total);for(int i=0;i<values.length;i++){n-=weights[i];if(n<0)return values[i];}return values[values.length-1];}
    private static long add(Timeline t,long at,Pt pos,String label,String type){
        if(!t.frames.isEmpty() && at<=t.frames.get(t.frames.size()-1).atMillis)at=t.frames.get(t.frames.size()-1).atMillis+100;
        t.frames.add(new Keyframe(at,pos,label,type));return at;
    }
    public static Timeline generate(Plan p,Config c){
        if(p.legs.isEmpty())throw new IllegalArgumentException("Нет участков маршрута");
        Timeline t=new Timeline();Random r=new Random(p.seed);
        long now=0;Pt last=null;
        for(int j=0;j<p.legs.size();j++){
            Leg l=p.legs.get(j);if(l.path.isEmpty())continue;
            Pt start=l.path.get(0),finish=l.path.get(l.path.size()-1);
            if(last!=null && metres(last,start)>35){throw new IllegalArgumentException("Разрыв маршрута между участками "+j+" и "+(j+1)+": добавьте соединяющий участок");}
            if(t.frames.isEmpty())now=add(t,0,start,"Отправление",l.type);
            boolean transit=publicTransport(l.type);
            if(transit && c.pauses){
                String waitName=rail(l.type)?"Ожидание поезда":"Ожидание транспорта";
                int wait=rail(l.type)?weighted(r,new int[]{30,60,90,120,180,240},new int[]{12,25,29,19,11,4}):weighted(r,new int[]{45,90,150,240,360,540},new int[]{12,24,29,20,11,4});
                if(j>0 && !WALK.equals(p.legs.get(j-1).type)){
                    int transfer=rail(l.type)?between(r,90,330):between(r,40,180);
                    now=add(t,now+1000L*transfer,start,"Пересадка",WAIT);t.events.add("Пересадка: "+transfer+" с");
                }
                now=add(t,now+wait*1000L,start,waitName,WAIT);t.events.add(waitName+": "+wait+" с");
            }
            double distance=length(l.path);
            int duration=l.baseSeconds;
            if(distance>5){
                double typical= WALK.equals(l.type)?between(r,37,55)/10.0/3.6 : CAR.equals(l.type)?between(r,24,52)/3.6 : rail(l.type)?between(r,24,51)/3.6 : between(r,13,33)/3.6;
                // Use base duration from router for roads and public transport; cap stochastic variation.
                double factor=WALK.equals(l.type)?0.84+r.nextDouble()*0.32 : (CAR.equals(l.type)?0.80+r.nextDouble()*0.4:0.94+r.nextDouble()*0.12);
                duration=Math.max(1,(int)Math.round(duration*factor/Math.max(0.5,Math.min(2,c.pace))));
                if(l.inferredGeometry && !CAR.equals(l.type))duration=Math.max(duration,(int)(distance/typical));
                if(c.trafficModel && (CAR.equals(l.type)||BUS.equals(l.type)||ELECTROBUS.equals(l.type)||MINIBUS.equals(l.type)||TROLLEY.equals(l.type)||TRAM.equals(l.type))){
                    // A statistical time-of-day model, not realtime congestion measurements.
                    java.time.ZoneId zone=java.time.ZoneId.systemDefault();
                    int hour=java.time.LocalTime.now(zone).getHour();
                    double scale=(hour>=7&&hour<10)||(hour>=17&&hour<20) ? 1.22+r.nextDouble()*0.53
                        :(hour>=22||hour<6 ? 0.95+r.nextDouble()*0.13:1.04+r.nextDouble()*0.29);
                    duration=(int)Math.max(1,Math.round(duration*scale));
                }
            }
            List<Pt> path=l.path;
            if(path.size()==1){now=add(t,now+duration*1000L,finish,l.name,l.type);last=finish;continue;}
            double[] cum=new double[path.size()];for(int i=1;i<path.size();i++)cum[i]=cum[i-1]+metres(path.get(i-1),path.get(i));
            ArrayList<Double> breakAt=new ArrayList<>();
            if(c.pauses && c.density>0 && distance>100){
                if(!rail(l.type)){
                    double step=WALK.equals(l.type)?260:(CAR.equals(l.type)?410:700);
                    for(double d=step;d<distance-step/2;d+=step){
                        if(chance(r,c.density==2?0.50:0.27))breakAt.add(d+between(r,-45,45));
                    }
                }
            }
            // Transit stop coordinates, when supplied by route provider, have priority over synthetic events.
            if(transit&&!l.stops.isEmpty()){
                breakAt.clear();for(Pt stop:l.stops){
                    double nearest=0,err=Double.MAX_VALUE;
                    for(int i=0;i<path.size();i++){double e=metres(path.get(i),stop);if(e<err){err=e;nearest=cum[i];}}
                    if(nearest>15&&nearest<distance-15&&err<200)breakAt.add(nearest);
                }
            }
            java.util.Collections.sort(breakAt);
            ArrayList<Double> markers=new ArrayList<>();markers.add(0.0);
            for(double d:breakAt){if(d>markers.get(markers.size()-1)+20 && d<distance-15) markers.add(d);}
            markers.add(distance);
            // Route in short geometric hops. Acceleration/easing around stops but no jumps.
            for(int part=1;part<markers.size();part++){
                double from=markers.get(part-1),to=markers.get(part);
                int n=Math.max(1,(int)Math.ceil((to-from)/25));
                for(int k=1;k<=n;k++){
                    double d=from+(to-from)*k/n;
                    int idx=1;while(idx<cum.length-1&&cum[idx]<d)idx++;
                    double span=Math.max(0.001,cum[idx]-cum[idx-1]);Pt point=mix(path.get(idx-1),path.get(idx),(d-cum[idx-1])/span);
                    double fraction=(to-from)/Math.max(0.001,distance)/n;
                    // Easing adjusts each hop duration by a small factor, preserving route geometry.
                    double phase=(double)k/n;
                    double easing=(k==1||k==n)?1.35:(phase<0.18||phase>0.82?1.14:0.91);
                    long hop=Math.max(250,Math.round(duration*1000*fraction*easing));
                    now=add(t,now+hop,point,l.name,l.type);
                }
                if(part<markers.size()-1 && c.pauses){
                    boolean stop=transit&&!l.stops.isEmpty();
                    String text=stop?"Остановка транспорта":WALK.equals(l.type)?"Пешеходный переход":CAR.equals(l.type)?"Светофор / затор":"Дорожная остановка";
                    int delay=stop?between(r,14,42):WALK.equals(l.type)?weighted(r,new int[]{10,20,35,50,70},new int[]{15,29,30,20,6}):CAR.equals(l.type)?weighted(r,new int[]{10,25,40,60,85},new int[]{15,25,28,24,8}):between(r,10,35);
                    Pt point=t.frames.get(t.frames.size()-1).pos;
                    now=add(t,now+delay*1000L,point,text,WAIT);t.events.add(text+": "+delay+" с");
                }
            }
            last=finish;
        }
        if(t.frames.size()<2)throw new IllegalArgumentException("Слишком короткий маршрут");
        return t;
    }
    public static Sample sample(Timeline t,long timeMillis,boolean noise,long seed){
        if(t.frames.isEmpty())throw new IllegalArgumentException("Пустой сценарий");
        int low=0,high=t.frames.size()-1;
        while(low<high){int m=(low+high+1)/2;if(t.frames.get(m).atMillis<=timeMillis)low=m;else high=m-1;}
        Keyframe a=t.frames.get(low),b=t.frames.get(Math.min(low+1,t.frames.size()-1));
        double fraction=b.atMillis<=a.atMillis?0:Math.max(0,Math.min(1,(timeMillis-a.atMillis)/(double)(b.atMillis-a.atMillis)));
        Pt p=mix(a.pos,b.pos,fraction);
        double secs=(b.atMillis-a.atMillis)/1000.0;
        double speed=secs<=0?0:Math.min(45,metres(a.pos,b.pos)/secs);
        if(timeMillis>=t.duration())speed=0;
        double bearing=metres(a.pos,b.pos)>0.5?bearing(a.pos,b.pos):0;
        String label=(metres(a.pos,b.pos)<0.2&&b.atMillis>a.atMillis)?b.label:a.label;
        String type=(metres(a.pos,b.pos)<0.2&&b.atMillis>a.atMillis)?WAIT:a.type;
        // Stable deterministic low-frequency GPS drift; does not imply actual movement.
        if(noise){double ts=timeMillis/1000.0;double north=Math.sin(ts*0.29+seed*.001)*1.7+Math.sin(ts*.071)*.7;
            double east=Math.cos(ts*.21+seed*.001)*1.5;
            p=new Pt(p.lat+north/111111.0,p.lon+east/(111111.0*Math.max(0.1,Math.cos(Math.toRadians(p.lat)))));
        }
        return new Sample(p,speed,bearing,label,type);
    }
}

package app.organicmaps.routeghost;

import org.json.JSONArray;
import org.json.JSONObject;
import org.xmlpull.v1.XmlPullParser;
import org.xmlpull.v1.XmlPullParserFactory;

import java.io.ByteArrayInputStream;
import java.nio.charset.StandardCharsets;
import java.util.ArrayList;
import java.util.List;
import java.util.Locale;
import static app.organicmaps.routeghost.RouteData.*;

/** Imported GPX is treated as user-supplied geometry, not a validated transit route. */
public final class TrackIO {
    private TrackIO(){}
    public static Plan read(byte[] bytes,String filename)throws Exception{
        String text=new String(bytes,StandardCharsets.UTF_8);
        if(filename.toLowerCase(Locale.ROOT).endsWith(".gpx")||text.trim().startsWith("<?xml"))return gpx(bytes);
        JSONObject json=new JSONObject(text);
        if(json.has("legs"))return Plan.read(json);
        JSONArray coords=null;
        if("FeatureCollection".equals(json.optString("type"))){JSONArray feats=json.getJSONArray("features");for(int j=0;j<feats.length();j++){JSONObject g=feats.getJSONObject(j).optJSONObject("geometry");if(g!=null&&"LineString".equals(g.optString("type"))){coords=g.optJSONArray("coordinates");break;}}}
        else if("Feature".equals(json.optString("type")))coords=json.getJSONObject("geometry").optJSONArray("coordinates");
        else if("LineString".equals(json.optString("type")))coords=json.optJSONArray("coordinates");
        if(coords==null)throw new Exception("Импортируются RouteGhost JSON, GeoJSON LineString или GPX");
        ArrayList<Pt> points=new ArrayList<>();for(int i=0;i<coords.length();i++){JSONArray p=coords.getJSONArray(i);points.add(new Pt(p.getDouble(1),p.getDouble(0)));}
        return make(points,"Импорт GeoJSON");
    }
    private static Plan gpx(byte[] data)throws Exception{
        XmlPullParser xml=XmlPullParserFactory.newInstance().newPullParser();xml.setInput(new ByteArrayInputStream(data),"UTF-8");
        ArrayList<Pt> points=new ArrayList<>();int event;
        while((event=xml.next())!=XmlPullParser.END_DOCUMENT){if(event==XmlPullParser.START_TAG && ("trkpt".equals(xml.getName())||"rtept".equals(xml.getName()))){
            points.add(new Pt(Double.parseDouble(xml.getAttributeValue(null,"lat")),Double.parseDouble(xml.getAttributeValue(null,"lon"))));
            if(points.size()>300_000)throw new Exception("GPX слишком большой");
        }}
        return make(points,"Импорт GPX");
    }
    private static Plan make(List<Pt> list,String name)throws Exception{
        if(list.size()<2)throw new Exception("В файле менее двух точек");
        Plan plan=new Plan();plan.title=name;plan.source="Пользовательский файл";
        plan.legs.add(new Leg(WALK,"Загруженный трек",list,Math.max(1,(int)Math.round(length(list)/1.35))));return plan;
    }
    public static String gpx(Plan p){
        StringBuilder s=new StringBuilder("<?xml version=\"1.0\" encoding=\"UTF-8\"?><gpx version=\"1.1\" creator=\"RouteGhost\" xmlns=\"http://www.topografix.com/GPX/1/1\"><trk><name>RouteGhost</name><trkseg>");
        for(Leg l:p.legs)for(Pt pt:l.path)s.append(String.format(Locale.US,"<trkpt lat=\"%.7f\" lon=\"%.7f\"/>",pt.lat,pt.lon));
        return s.append("</trkseg></trk></gpx>").toString();
    }
}
